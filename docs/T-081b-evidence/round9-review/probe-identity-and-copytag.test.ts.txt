// round9 independent review — throwaway probes (deleted after the run; output kept)
import { describe, it, expect } from "vitest";
import { validateThemeBundle } from "../themeBundle";
import { makeMessages } from "../../i18n/compose";
import { zh } from "../../i18n/locales/zh";
import { applyWording } from "../../i18n/wording";
import { MESSAGE_KEYS } from "../../i18n/messageKeys.generated";

describe("PROBE 1 — a pack purpose-built to overwrite themeIdentity", () => {
  it("reports what survives validation", () => {
    const b: any = {
      id: "attack9", name: "辦公室", colors: { "--color-accent": "#0b1020" },
      wording: { zh: {
        ...Object.fromEntries(MESSAGE_KEYS.map((k) => [k, "偽造"])),
        "themeIdentity.office": "精靈村",
        "themeIdentity.newTheme": "精靈村",
        "profile.themeOffice": "精靈村",
        "themeMarkers.builtinGroup": "自訂",
      } },
    };
    const err = validateThemeBundle(b);
    const k = Object.keys(b.wording.zh);
    console.log("verdict:", err);
    console.log("themeIdentity.office survives? ", k.includes("themeIdentity.office"));
    console.log("themeIdentity.newTheme survives?", k.includes("themeIdentity.newTheme"));
    console.log("profile.themeOffice survives?   ", k.includes("profile.themeOffice"));
    console.log("themeMarkers.builtinGroup survives?", k.includes("themeMarkers.builtinGroup"),
                "value =", b.wording.zh["themeMarkers.builtinGroup"]);
    expect(err).toBeNull();
    expect(k.includes("themeIdentity.office")).toBe(false);
  });
});

describe("PROBE 2 — a pack stretches themeMarkers.copyTag, then the BUILT-IN theme is downloaded", () => {
  it("reports whether the resulting file re-imports", () => {
    const overlay = { "themeMarkers.copyTag": "很長的副本標記".repeat(20) };
    const packErr = validateThemeBundle({ id: "longcopy", name: "長副本包",
      colors: { "--color-bg": "#101018" }, wording: { zh: overlay } } as any);
    console.log("pack accepted?", packErr);
    const name = makeMessages(applyWording(zh, overlay) as any, "zh")
      .themeCopyName(zh.themeIdentity.office);
    console.log("built-in row's download name length:", name.length);
    console.log("re-import verdict:",
      validateThemeBundle({ id: "office-base", name, colors: { "--color-bg": "#101018" } }));
    expect(packErr).toBeNull();
  });
});
